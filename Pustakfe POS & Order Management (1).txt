const playNotificationSound = (type = 'swiggy') => {
  try {
    const AudioContext = window.AudioContext || window.webkitAudioContext;
    if (!AudioContext) return;
    const ctx = new AudioContext();

    if (type === 'swiggy') {
      const osc1 = ctx.createOscillator();
      const gain1 = ctx.createGain();
      osc1.type = 'sine';
      osc1.frequency.setValueAtTime(587.33, ctx.currentTime);
      osc1.frequency.setValueAtTime(880, ctx.currentTime + 0.15);
      gain1.gain.setValueAtTime(0.3, ctx.currentTime);
      gain1.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.6);
      osc1.connect(gain1);
      gain1.connect(ctx.destination);
      osc1.start();
      osc1.stop(ctx.currentTime + 0.6);
    } else {
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(783.99, ctx.currentTime);
      osc.frequency.setValueAtTime(1046.50, ctx.currentTime + 0.12);
      gain.gain.setValueAtTime(0.35, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.5);
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start();
      osc.stop(ctx.currentTime + 0.5);
    }
  } catch (err) {
    console.warn('Audio playback error', err);
  }
};

const loadStorage = (key, fallback) => {
  try {
    const saved = localStorage.getItem(key);
    return saved ? JSON.parse(saved) : fallback;
  } catch (err) {
    return fallback;
  }
};

const saveStorage = (key, data) => {
  try {
    localStorage.setItem(key, JSON.stringify(data));
  } catch (err) {
    console.error(`Error saving ${key}`, err);
  }
};

      {showCategoryModal && (
        <div className="fixed inset-0 bg-black/80 z-50 flex items-center justify-center p-4 backdrop-blur-sm">
          <div className="bg-slate-900 border border-slate-700 text-white rounded-xl max-w-xs w-full p-4 shadow-2xl">
            <h3 className="font-bold text-sm mb-3 flex items-center gap-1.5">
              <FolderPlus className="w-4 h-4 text-amber-400" />
              <span>Nayi Category Add Karein</span>
            </h3>

            <form onSubmit={handleAddCategory} className="space-y-3 text-xs">
              <input
                type="text"
                autoFocus
                required
                placeholder="e.g. Artisanal Breads, Desserts..."
                value={newCatName}
                onChange={(e) => setNewCatName(e.target.value)}
                className="w-full bg-slate-950 border border-slate-700 rounded-lg p-2 text-white focus:outline-none focus:border-amber-500"
              />

              <div className="flex justify-end space-x-2">
                <button
                  type="button"
                  onClick={() => setShowCategoryModal(false)}
                  className="bg-slate-800 text-slate-300 px-3 py-1.5 rounded-lg"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className="bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold px-3 py-1.5 rounded-lg"
                >
                  Create
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Thermal KOT Slip Modal with Pustakfe Branding */}
      {kotModalData && (
        <div className="fixed inset-0 bg-black/80 z-50 flex items-center justify-center p-4 backdrop-blur-sm">
          <div className="bg-white text-slate-900 rounded-lg max-w-xs w-full p-4 shadow-2xl font-mono text-xs">
            <div className="text-center pb-2 border-b border-dashed border-slate-400">
              <h2 className="font-black text-sm tracking-widest uppercase">--- KITCHEN ORDER TICKET ---</h2>
              <div className="text-[12px] font-black mt-1 uppercase tracking-wider text-slate-900">pustakfe Books & Café</div>
              <div className="flex justify-between mt-1 text-[10px] text-slate-600">
                <span>KOT: {kotModalData.id}</span>
                <span>{kotModalData.time}</span>
              </div>
              <div className="text-[12px] font-bold mt-1 bg-slate-100 py-0.5">
                {kotModalData.table} ({kotModalData.type})
              </div>
            </div>

            <div className="py-3 border-b border-dashed border-slate-400 space-y-2">
              {kotModalData.items.map((item, i) => (
                <div key={i} className="flex justify-between items-start">
                  <div>
                    <span className="font-bold">{item.name}</span>
                    {item.variantName && (
                      <div className="text-[10px] text-amber-700 font-bold">
                        [{item.variantName}]
                      </div>
                    )}
                    {item.addons && item.addons.length > 0 && (
                      <div className="text-[10px] text-emerald-700 font-semibold">
                        + {item.addons.join(', ')}
                      </div>
                    )}
                  </div>
                  <span className="font-black text-sm">x{item.qty}</span>
                </div>
              ))}
            </div>

            <div className="mt-4 flex space-x-2">
              <button
                onClick={() => setKotModalData(null)}
                className="flex-1 bg-slate-200 hover:bg-slate-300 text-slate-800 font-bold py-1.5 rounded text-xs"
              >
                Close
              </button>
              <button
                onClick={() => {
                  window.print();
                  setKotModalData(null);
                }}
                className="flex-1 bg-slate-900 hover:bg-slate-800 text-white font-bold py-1.5 rounded text-xs flex items-center justify-center space-x-1"
              >
                <Printer className="w-3.5 h-3.5" />
                <span>Print KOT</span>
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Bill & Settlement Modal with Pustakfe GST Tax Invoice */}
      {billModalData && (
        <div className="fixed inset-0 bg-black/80 z-50 flex items-center justify-center p-4 backdrop-blur-sm">
          <div className="bg-white text-slate-900 rounded-lg max-w-sm w-full p-5 shadow-2xl font-mono text-xs max-h-[95vh] flex flex-col justify-between">
            <div className="overflow-y-auto pr-1">
              <div className="text-center pb-2 border-b border-dashed border-slate-400">
                <h1 className="font-black text-lg tracking-wide lowercase">pustakfe</h1>
                <p className="text-[10px] font-semibold text-slate-600 uppercase">Books & Café • Navi Mumbai</p>
                <p className="text-[9px] text-slate-500">Sector 17, Vashi • GSTIN: 27AABCP8921P1Z5</p>
                <div className="flex justify-between mt-2 pt-1 border-t border-slate-200 text-[10px]">
                  <span>Bill: <strong>{billModalData.invoiceNo}</strong></span>
                  <span>{billModalData.date} • {billModalData.time}</span>
                </div>
              </div>

              <div className="py-2 border-b border-dashed border-slate-400 space-y-1 text-slate-700">
                <div className="flex justify-between">
                  <span>Subtotal:</span>
                  <span>₹{billModalData.subtotal.toFixed(2)}</span>
                </div>
                {billModalData.discountAmount > 0 && (
                  <div className="flex justify-between text-amber-700 font-bold">
                    <span>Discount ({billModalData.discountPercent}%):</span>
                    <span>- ₹{billModalData.discountAmount.toFixed(2)}</span>
                  </div>
                )}
                <div className="flex justify-between">
                  <span>CGST (2.5%) + SGST (2.5%):</span>
                  <span>₹{(billModalData.cgst + billModalData.sgst).toFixed(2)}</span>
                </div>
                <div className="flex justify-between font-black text-sm text-slate-900 pt-1 border-t border-slate-300">
                  <span>TOTAL PAYABLE:</span>
                  <span>₹{billModalData.grandTotal}</span>
                </div>
              </div>

              <div className="py-2 text-[10px] text-center text-slate-500 italic">
                Thank you for visiting pustakfe! Happy Reading & Sipping ☕📚
              </div>
            </div>

            <div className="mt-3 pt-3 border-t border-slate-200">
              <span className="text-[11px] font-bold text-slate-700 block mb-1.5">Settle Payment & Free Table:</span>
              <div className="grid grid-cols-3 gap-2">
                <button
                  onClick={() => handleSettleBill('Cash')}
                  className="bg-emerald-600 hover:bg-emerald-500 text-white font-bold py-2 rounded flex flex-col items-center justify-center text-[11px]"
                >
                  <Banknote className="w-4 h-4 mb-0.5" />
                  <span>Cash</span>
                </button>
                <button
                  onClick={() => handleSettleBill('Card')}
                  className="bg-blue-600 hover:bg-blue-500 text-white font-bold py-2 rounded flex flex-col items-center justify-center text-[11px]"
                >
                  <CreditCard className="w-4 h-4 mb-0.5" />
                  <span>Card</span>
                </button>
                <button
                  onClick={() => handleSettleBill('UPI / QR')}
                  className="bg-amber-600 hover:bg-amber-500 text-white font-bold py-2 rounded flex flex-col items-center justify-center text-[11px]"
                >
                  <QrCode className="w-4 h-4 mb-0.5" />
                  <span>UPI / QR</span>
                </button>
              </div>

              <button
                onClick={() => setBillModalData(null)}
                className="w-full mt-2 bg-slate-200 hover:bg-slate-300 text-slate-800 font-bold py-1.5 rounded text-xs"
              >
                Cancel / Edit Order
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
